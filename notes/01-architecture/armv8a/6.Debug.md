**Debug Host**
运行调试器的电脑，可以发送high-level调试命令比如设置一个断点或查看一块内存的内容。
**Protocol converter**
Debug Host通过以太网接口发送消息给Debug target，但是Debug target实现的是不同的接口协议，就需要协议转换器转换两种协议。
**Debug target**
支持协议转换器访问debug unit的最底层系统。
**Debug unit**
帮助调试core上的软件：
* 基于core的硬件系统
* 操作系统
* 应用软件
可实现：
* 停止程序执行
* 检查和改变处理器和协处理器状态
* 检查和改变内存，以及外围设备输入输出状态
* restart core

## 内核侧调试实现总览

上面的架构图描述的是**外部调试**（JTAG/SWD 探针 + 调试主机）。Linux 更常用的是**自托管调试**（self-hosted debug）：CPU 自己执行断点/观察点逻辑，异常回到内核，内核再通知 ptrace 或 perf。arm64 的自托管调试由三个文件分工：

| 文件 | 职责 |
| :-- | :-- |
| `arch/arm64/kernel/debug-monitors.c` | MDSCR_EL1 管理、调试监视器使能、单步 API、BRK 指令 hook |
| `arch/arm64/kernel/hw_breakpoint.c` | 断点/观察点寄存器编程、perf 集成、异常处理 |
| `arch/arm64/kernel/ptrace.c` | `NT_ARM_HW_BREAK`/`NT_ARM_HW_WATCH` regset、ptrace→perf 桥接 |

本树 `.config`：`CONFIG_HAVE_HW_BREAKPOINT=y`、`CONFIG_PERF_EVENTS=y`、`CONFIG_DEBUG_INFO=y`，但 `CONFIG_KGDB` **未启用**、`CONFIG_CORESIGHT` **未启用**、`CONFIG_KPROBES`/`CONFIG_FTRACE` 未启用。因此下文 kgdb 与 CoreSight 部分均属"源码存在，当前配置未启用"。

编译开关见 `arch/arm64/kernel/Makefile`：

```make
obj-y			:= debug-monitors.o entry.o irq.o fpsimd.o ...
obj-$(CONFIG_HAVE_HW_BREAKPOINT)	+= hw_breakpoint.o
obj-$(CONFIG_KGDB)			+= kgdb.o
```

`debug-monitors.o` 无条件编译（单步与 BRK 处理是内核自身需要的），`hw_breakpoint.o` 与 `kgdb.o` 按配置裁剪。

## 调试寄存器与 MDSCR_EL1

### DBGBVR/DBGBCR/DBGWVR/DBGWCR

arm64 的硬件断点/观察点由四组寄存器实现：

| 寄存器 | 全称 | 作用 |
| :-- | :-- | :-- |
| `DBGBVR<n>_EL1` | Breakpoint Value Register | 断点地址（指令地址） |
| `DBGBCR<n>_EL1` | Breakpoint Control Register | 使能、特权级、字节选择 |
| `DBGWVR<n>_EL1` | Watchpoint Value Register | 观察点地址（数据地址） |
| `DBGWCR<n>_EL1` | Watchpoint Control Register | 使能、特权级、读/写类型、字节选择 |

数量由 `ID_AA64DFR0_EL1` 的 BRPs/WRPs 字段决定，`arch/arm64/include/asm/hw_breakpoint.h:139`：

```c
/* Determine number of BRP registers available. */
static inline int get_num_brps(void)
{
	u64 dfr0 = read_sanitised_ftr_reg(SYS_ID_AA64DFR0_EL1);
	return 1 +
		cpuid_feature_extract_unsigned_field(dfr0,
						ID_AA64DFR0_EL1_BRPs_SHIFT);
}

/* Determine number of WRP registers available. */
static inline int get_num_wrps(void)
{
	u64 dfr0 = read_sanitised_ftr_reg(SYS_ID_AA64DFR0_EL1);
	return 1 +
		cpuid_feature_extract_unsigned_field(dfr0,
						ID_AA64DFR0_EL1_WRPs_SHIFT);
}
```

注意 **`+1`**：ID 寄存器里编码的是"数量减一"。前文架构图提到 Cortex-A55 有 6 个断点、4 个观察点，对应 BRPs=5、WRPs=3。架构上限是 16 个（`ARM_MAX_BRP`/`ARM_MAX_WRP`，`arch/arm64/include/asm/hw_breakpoint.h:82`）。

控制寄存器的位布局被抽象成位域结构（`arch/arm64/include/asm/hw_breakpoint.h:13`）：

```c
struct arch_hw_breakpoint_ctrl {
	u32 __reserved	: 19,
		len		: 8,
		type		: 2,
		privilege	: 2,
		enabled		: 1;
};

struct arch_hw_breakpoint {
	u64 address;
	u64 trigger;
	struct arch_hw_breakpoint_ctrl ctrl;
};
```

编码/解码函数（`arch/arm64/include/asm/hw_breakpoint.h:33`）：

```c
static inline u32 encode_ctrl_reg(struct arch_hw_breakpoint_ctrl ctrl)
{
	u32 val = (ctrl.len << 5) | (ctrl.type << 3) | (ctrl.privilege << 1) |
		ctrl.enabled;

	if (is_kernel_in_hyp_mode() && ctrl.privilege == AARCH64_BREAKPOINT_EL1)
		val |= DBG_HMC_HYP;

	return val;
}

static inline void decode_ctrl_reg(u32 reg,
				   struct arch_hw_breakpoint_ctrl *ctrl)
{
	ctrl->enabled	= reg & 0x1;
	reg >>= 1;
	ctrl->privilege	= reg & 0x3;
	reg >>= 2;
	ctrl->type	= reg & 0x3;
	reg >>= 2;
	ctrl->len	= reg & 0xff;
}
```

- `len` 是**字节选择掩码**（BAS），不是长度数值。`ARM_BREAKPOINT_LEN_4 = 0xf` 表示低 4 字节有效。断点用它选择 4 字节指令槽中的哪几个字节；观察点用它选择 8 字节窗口内的哪些字节。
- `type`：0 = 执行（断点），1 = load，2 = store（观察点）。
- `privilege`：1 = EL1，2 = EL0。
- `DBG_HMC_HYP`（bit 13）在 VHE 模式下额外允许 EL2 命中，因为此时内核跑在 EL2。

### 寄存器读写：read_wb_reg/write_wb_reg

系统寄存器名不能用变量索引，所以驱动用宏展开出 16×4 个 case（`arch/arm64/kernel/hw_breakpoint.c:107`）：

```c
static u64 read_wb_reg(int reg, int n)
{
	u64 val = 0;

	switch (reg + n) {
	GEN_READ_WB_REG_CASES(AARCH64_DBG_REG_BVR, AARCH64_DBG_REG_NAME_BVR, val);
	GEN_READ_WB_REG_CASES(AARCH64_DBG_REG_BCR, AARCH64_DBG_REG_NAME_BCR, val);
	GEN_READ_WB_REG_CASES(AARCH64_DBG_REG_WVR, AARCH64_DBG_REG_NAME_WVR, val);
	GEN_READ_WB_REG_CASES(AARCH64_DBG_REG_WCR, AARCH64_DBG_REG_NAME_WCR, val);
	default:
		pr_warn("attempt to read from unknown breakpoint register %d\n", n);
	}

	return val;
}
NOKPROBE_SYMBOL(read_wb_reg);

static void write_wb_reg(int reg, int n, u64 val)
{
	switch (reg + n) {
	GEN_WRITE_WB_REG_CASES(...);
	default:
		pr_warn("attempt to write to unknown breakpoint register %d\n", n);
	}
	isb();
}
NOKPROBE_SYMBOL(write_wb_reg);
```

虚拟寄存器基址按 16 个一组排布（`arch/arm64/include/asm/hw_breakpoint.h:86`）：

```c
#define AARCH64_DBG_REG_BVR	0
#define AARCH64_DBG_REG_BCR	(AARCH64_DBG_REG_BVR + ARM_MAX_BRP)
#define AARCH64_DBG_REG_WVR	(AARCH64_DBG_REG_BCR + ARM_MAX_BRP)
#define AARCH64_DBG_REG_WCR	(AARCH64_DBG_REG_WVR + ARM_MAX_WRP)
```

两个细节：写后必须 `isb()` 保证调试寄存器生效；两个函数都标 `NOKPROBE_SYMBOL`，防止 kprobe 递归（本树 kprobe 未启用，但标注保留）。

### MDSCR_EL1 与调试监视器使能

`MDSCR_EL1`（Monitor Debug System Control Register）是自托管调试的总开关，关键位：

| 位 | 名称 | 作用 |
| :-- | :-- | :-- |
| 0 | SS | Software Step，使能单步 |
| 13 | KDE | Kernel Debug Enable，允许 EL1 调试异常 |
| 15 | MDE | Monitor Debug Enable，允许 EL0 调试异常 |
| 12 | TDCC |  trapping DCC 访问 |

访问封装在 `arch/arm64/kernel/debug-monitors.c:41`：

```c
static void mdscr_write(u64 mdscr)
{
	unsigned long flags;
	flags = local_daif_save();
	write_sysreg(mdscr, mdscr_el1);
	local_daif_restore(flags);
}
NOKPROBE_SYMBOL(mdscr_write);

static u64 mdscr_read(void)
{
	return read_sysreg(mdscr_el1);
}
```

写之前关 DAIF，避免读-改-写窗口被调试异常打断。

使能/禁用用 per-CPU 引用计数管理（`arch/arm64/kernel/debug-monitors.c:84`）：

```c
static DEFINE_PER_CPU(int, mde_ref_count);
static DEFINE_PER_CPU(int, kde_ref_count);

void enable_debug_monitors(enum dbg_active_el el)
{
	u64 mdscr, enable = 0;

	WARN_ON(preemptible());

	if (this_cpu_inc_return(mde_ref_count) == 1)
		enable = MDSCR_EL1_MDE;

	if (el == DBG_ACTIVE_EL1 &&
	    this_cpu_inc_return(kde_ref_count) == 1)
		enable |= MDSCR_EL1_KDE;

	if (enable && debug_enabled) {
		mdscr = mdscr_read();
		mdscr |= enable;
		mdscr_write(mdscr);
	}
}
```

引用计数是必要的：多个断点可能同时安装，只有最后一个卸载时才真正关掉 MDE/KDE。`debug_enabled` 是一个 debugfs 布尔（`/sys/kernel/debug/debug_enabled`），也可用启动参数 `nodebugmon` 关闭——注释说明这是为了**让外部 JTAG 调试器接管**，对应前文架构图里的 Debug Host 路径。

### OS Lock

`arch/arm64/kernel/debug-monitors.c:129`：

```c
static int clear_os_lock(unsigned int cpu)
{
	write_sysreg(0, osdlr_el1);
	write_sysreg(0, oslar_el1);
	isb();
	return 0;
}

static int __init debug_monitors_init(void)
{
	return cpuhp_setup_state(CPUHP_AP_ARM64_DEBUG_MONITORS_STARTING,
				 "arm64/debug_monitors:starting",
				 clear_os_lock, NULL);
}
postcore_initcall(debug_monitors_init);
```

OS Lock 是架构提供的"调试单元锁"，锁住时自托管调试寄存器访问被忽略。每个 CPU 上线时必须解锁，否则断点写不进去。`cpu_do_resume` 里也会恢复 OSLAR（见 `arch/arm64/mm/proc.S`）。

## 硬件断点/观察点的 perf 抽象

Linux 不把断点当成 ptrace 专属功能，而是统一建模为 **perf 事件**（`PERF_TYPE_BREAKPOINT`）。这样 gdb、perf、内核自身的 kprobe 都能共享同一套槽位管理。

### 槽位管理

`arch/arm64/kernel/hw_breakpoint.c:32`：

```c
/* Breakpoint currently in use for each BRP. */
static DEFINE_PER_CPU(struct perf_event *, bp_on_reg[ARM_MAX_BRP]);

/* Watchpoint currently in use for each WRP. */
static DEFINE_PER_CPU(struct perf_event *, wp_on_reg[ARM_MAX_WRP]);

/* Currently stepping a per-CPU kernel breakpoint. */
static DEFINE_PER_CPU(int, stepping_kernel_bp);

/* Number of BRP/WRP registers on this CPU. */
static int core_num_brps;
static int core_num_wrps;
```

per-CPU 数组记录"哪个硬件槽位被哪个 perf 事件占用"。`hw_breakpoint_slots()`（`arch/arm64/kernel/hw_breakpoint.c:44`）向 perf 核心报告可用槽位数，perf 据此做全局调度——任务迁移时把事件从旧 CPU 卸载、在新 CPU 安装。

### arch_install_hw_breakpoint

`hw_breakpoint_control()`（`arch/arm64/kernel/hw_breakpoint.c:224`）是核心：

```c
	struct arch_hw_breakpoint *info = counter_arch_bp(bp);
	struct perf_event **slots;
	struct debug_info *debug_info = &current->thread.debug;
	int i, max_slots, ctrl_reg, val_reg, reg_enable;
	enum dbg_active_el dbg_el = debug_exception_level(info->ctrl.privilege);
	u32 ctrl;

	if (info->ctrl.type == ARM_BREAKPOINT_EXECUTE) {
		/* Breakpoint */
		ctrl_reg = AARCH64_DBG_REG_BCR;
		val_reg = AARCH64_DBG_REG_BVR;
		slots = this_cpu_ptr(bp_on_reg);
		max_slots = core_num_brps;
		reg_enable = !debug_info->bps_disabled;
	} else {
		/* Watchpoint */
		ctrl_reg = AARCH64_DBG_REG_WCR;
		val_reg = AARCH64_DBG_REG_WVR;
		slots = this_cpu_ptr(wp_on_reg);
		max_slots = core_num_wrps;
		reg_enable = !debug_info->wps_disabled;
	}

	i = hw_breakpoint_slot_setup(slots, max_slots, bp, ops);

	if (WARN_ONCE(i < 0, "Can't find any breakpoint slot"))
		return i;

	switch (ops) {
	case HW_BREAKPOINT_INSTALL:
		enable_debug_monitors(dbg_el);
		fallthrough;
	case HW_BREAKPOINT_RESTORE:
		/* Setup the address register. */
		write_wb_reg(val_reg, i, info->address);

		/* Setup the control register. */
		ctrl = encode_ctrl_reg(info->ctrl);
		write_wb_reg(ctrl_reg, i,
			     reg_enable ? ctrl | 0x1 : ctrl & ~0x1);
		break;
	case HW_BREAKPOINT_UNINSTALL:
		write_wb_reg(ctrl_reg, i, 0);
		disable_debug_monitors(dbg_el);
		break;
	}
```

`reg_enable` 来自 `current->thread.debug.bps_disabled`/`wps_disabled`——这是"断点命中后临时禁用"的标志，见后文单步部分。安装时先开调试监视器再写寄存器，卸载时先清寄存器再关监视器，顺序不能反。

对外接口只有两个（`arch/arm64/kernel/hw_breakpoint.c:290`）：

```c
int arch_install_hw_breakpoint(struct perf_event *bp)
{
	return hw_breakpoint_control(bp, HW_BREAKPOINT_INSTALL);
}

void arch_uninstall_hw_breakpoint(struct perf_event *bp)
{
	hw_breakpoint_control(bp, HW_BREAKPOINT_UNINSTALL);
}
```

## ptrace 接口

### NT_ARM_HW_BREAK / NT_ARM_HW_WATCH

gdb 设置硬件断点走的是 `PTRACE_SETREGSET` + `NT_ARM_HW_BREAK`（观察点用 `NT_ARM_HW_WATCH`）。regset 定义见 [9.ABI](9.ABI.md) 的 ptrace 章节，其 `.regset_get`/`.set` 分别是 `hw_break_get`/`hw_break_set`。

用户态可见的布局是 `struct user_hwdebug_state`（`arch/arm64/include/uapi/asm/ptrace.h:102`）：

```c
struct user_hwdebug_state {
	__u32		dbg_info;
	__u32		pad;
	struct {
		__u64	addr;
		__u32	ctrl;
		__u32	pad;
	}		dbg_regs[16];
};
```

### ptrace_hbp_create：ptrace → perf

关键桥接在 `arch/arm64/kernel/ptrace.c:285`：

## 调试异常处理路径

### 向量与 EC 分派

调试异常是**同步异常**，走 sync 槽位，由 ESR_EL1.EC 区分：

| EC | 含义 | EL1 handler | EL0 handler |
| :-- | :-- | :-- | :-- |
| `ESR_ELx_EC_BREAKPT_CUR`/`_LOW` | 硬件断点 | `el1_breakpt` | `el0_breakpt` |
| `ESR_ELx_EC_WATCHPT_CUR`/`_LOW` | 观察点 | `el1_watchpt` | `el0_watchpt` |
| `ESR_ELx_EC_SOFTSTP_CUR`/`_LOW` | 软件单步 | `el1_softstp` | `el0_softstp` |
| `ESR_ELx_EC_BRK64` | BRK 指令 | `el1_brk64` | `el0_brk64` |

### do_breakpoint

`arch/arm64/kernel/hw_breakpoint.c:631`：

```c
	slots = this_cpu_ptr(bp_on_reg);
	addr = instruction_pointer(regs);
	debug_info = &current->thread.debug;

	for (i = 0; i < core_num_brps; ++i) {
		rcu_read_lock();

		bp = slots[i];
		if (bp == NULL)
			goto unlock;

		/* Check if the breakpoint value matches. */
		val = read_wb_reg(AARCH64_DBG_REG_BVR, i);
		if (val != (addr & ~0x3))
			goto unlock;

		/* Possible match, check the byte address select to confirm. */
		ctrl_reg = read_wb_reg(AARCH64_DBG_REG_BCR, i);
		decode_ctrl_reg(ctrl_reg, &ctrl);
		if (!((1 << (addr & 0x3)) & ctrl.len))
			goto unlock;

		counter_arch_bp(bp)->trigger = addr;
		perf_bp_event(bp, regs);

		/* Do we need to handle the stepping? */
		if (is_default_overflow_handler(bp))
			step = 1;
unlock:
		rcu_read_unlock();
	}
```

两级匹配：先比 4 字节对齐的地址，再用 BAS 掩码确认具体字节。这是因为 A64 指令固定 4 字节，但架构允许按字节设置断点。命中后 `perf_bp_event()` 通知 perf 核心（ptrace 场景下最终发 SIGTRAP）。

### do_watchpoint 与地址启发式

观察点比断点麻烦：硬件报告的 FAR 不一定精确落在被观察的字节上。`arch/arm64/kernel/hw_breakpoint.c:705` 的注释解释得很清楚：

```c
/*
 * Arm64 hardware does not always report a watchpoint hit address that matches
 * one of the watchpoints set. It can also report an address "near" the
 * watchpoint if a single instruction access both watched and unwatched
 * addresses. There is no straight-forward way, short of disassembling the
 * offending instruction, to map that address back to the watchpoint. This
 * function computes the distance of the memory access from the watchpoint as a
 * heuristic for the likelihood that a given access triggered the watchpoint.
 *
 * See Section D2.10.5 "Determining the memory location that caused a Watchpoint
 * exception" of ARMv8 Architecture Reference Manual for details.
 */
static u64 get_distance_from_watchpoint(unsigned long addr, u64 val,
					struct arch_hw_breakpoint_ctrl *ctrl)
```

`do_watchpoint()`（`arch/arm64/kernel/hw_breakpoint.c:762`）遍历所有 WRP，用 `watchpoint_report()` 计算距离，选最接近的那个报告。一条 `ldp`/`stp` 可能同时触碰被观察与未被观察的地址，硬件只给一个近似地址，内核只能启发式归因。

### 调用链

```
硬件断点/观察点命中
  → vectors: kernel_ventry 1, h, 64, sync（或 0, t, 64, sync）
  → el1h_64_sync_handler / el0t_64_sync_handler
  → 按 ESR_ELx_EC 分派
  → el1_breakpt / el1_watchpt / el1_softstp
      arm64_enter_el1_dbg()          // ct_nmi_enter
      debug_exception_enter()        // preempt_disable
  → do_breakpoint(esr, regs) / do_watchpoint(far, esr, regs)
  → 匹配 bp_on_reg[] / wp_on_reg[] 槽位
  → perf_bp_event(bp, regs)
  → ptrace_hbptriggered() → force_sig_fault(SIGTRAP, ...)
      或 perf 溢出处理 → 单步越过
  → debug_exception_exit() → arm64_exit_el1_dbg()
  → ret_to_user / ret_to_kernel
```

## 单步执行

arm64 有两种单步机制，分别用于用户态与内核态。

### 用户态单步（SPSR.SS）

`arch/arm64/kernel/debug-monitors.c:382`：

```c
/* ptrace API */

{
	struct thread_info *ti = task_thread_info(task);

	if (!test_and_set_ti_thread_flag(ti, TIF_SINGLESTEP))
		set_regs_spsr_ss(task_pt_regs(task));
}
NOKPROBE_SYMBOL(user_enable_single_step);

void user_disable_single_step(struct task_struct *task)
{
	clear_ti_thread_flag(task_thread_info(task), TIF_SINGLESTEP);
}
```

用户态单步靠 **SPSR.SS 位**（`DBG_SPSR_SS`）：`eret` 返回用户态时若 SS=1，执行一条指令后立刻产生 Software Step 异常。`TIF_SINGLESTEP` 标志让 `ret_to_user` 路径上的 `enable_step_tsk` 宏（`arch/arm64/kernel/entry.S`）在每次返回前重新置位 SPSR.SS。

### 内核态单步（MDSCR_EL1.SS）

`arch/arm64/kernel/debug-monitors.c:347`：

```c
/* Kernel API */
void kernel_enable_single_step(struct pt_regs *regs)
{
	WARN_ON(!irqs_disabled());
	set_regs_spsr_ss(regs);
	mdscr_write(mdscr_read() | MDSCR_EL1_SS);
	enable_debug_monitors(DBG_ACTIVE_EL1);
}
NOKPROBE_SYMBOL(kernel_enable_single_step);

void kernel_disable_single_step(void)
{
	WARN_ON(!irqs_disabled());
	mdscr_write(mdscr_read() & ~MDSCR_EL1_SS);
	disable_debug_monitors(DBG_ACTIVE_EL1);
}

int kernel_active_single_step(void)
{
	WARN_ON(!irqs_disabled());
	return mdscr_read() & MDSCR_EL1_SS;
}
```

内核态单步需要 **MDSCR_EL1.SS + KDE + SPSR.SS** 三者同时置位。`WARN_ON(!irqs_disabled())` 是因为单步状态是 per-CPU 的，抢占会破坏它。

### 断点后的"步进越过"

硬件断点命中后，PC 还停在断点指令上。若直接返回，会立刻再次命中，形成死循环。解法是：**临时禁用所有同类断点 → 单步执行一条指令 → 重新启用**。

`do_breakpoint()` 的后半段（`arch/arm64/kernel/hw_breakpoint.c:676`）：

```c
	if (user_mode(regs)) {
		debug_info->bps_disabled = 1;
		toggle_bp_registers(AARCH64_DBG_REG_BCR, DBG_ACTIVE_EL0, 0);

		/* If we're already stepping a watchpoint, just return. */
		if (debug_info->wps_disabled)
			return;

		if (test_thread_flag(TIF_SINGLESTEP))
			debug_info->suspended_step = 1;
		else
			user_enable_single_step(current);
	} else {
		toggle_bp_registers(AARCH64_DBG_REG_BCR, DBG_ACTIVE_EL1, 0);
		kernel_step = this_cpu_ptr(&stepping_kernel_bp);

		if (*kernel_step != ARM_KERNEL_STEP_NONE)
			return;

		if (kernel_active_single_step()) {
			*kernel_step = ARM_KERNEL_STEP_SUSPEND;
		} else {
			*kernel_step = ARM_KERNEL_STEP_ACTIVE;
			kernel_enable_single_step(regs);
		}
	}
```

`toggle_bp_registers()`（`arch/arm64/kernel/hw_breakpoint.c:591`）批量改 BCR/WCR 的 enable 位，只影响指定特权级的槽位。

单步完成后由 `try_step_suspended_breakpoints()`（`arch/arm64/kernel/hw_breakpoint.c:856`）恢复断点。`el1_softstp()` 里先调它，若返回 true 说明这次单步只是为了越过断点，不需要再通知调试器：

```c
		/*
		 * After handling a breakpoint, we suspend the breakpoint
		 * and use single-step to move to the next instruction.
		 * If we are stepping a suspended breakpoint there's nothing more to do:
		 * the single-step is complete.
		 */
		if (!try_step_suspended_breakpoints(regs))
			do_el1_softstep(esr, regs);
```

`stepping_kernel_bp` 的三态（`ARM_KERNEL_STEP_NONE`/`ACTIVE`/`SUSPEND`）处理"单步中又命中断点"的嵌套场景。

- **动态断点**：`kgdb_arch_set_breakpoint()`（`arch/arm64/kernel/kgdb.c:315`）用 `aarch64_insn_write()` 把目标指令替换成 BRK：

```c
	ret = perf_pmu_register(&etm_pmu, CORESIGHT_ETM_PMU_NAME, -1);
```

用户态用 `perf record -e cs_etm//` 即可采集指令追踪，数据通过 perf 的 AUX buffer 机制导出。`etm_setup_aux()` 里调用 `coresight_build_path()`（`drivers/hwtracing/coresight/coresight-etm-perf.c:407`）建立 source→sink 路径。`addr_filters` 支持按地址范围过滤追踪，减少数据量。

### TRBE

`drivers/hwtracing/coresight/coresight-trbe.c` 实现 FEAT_TRBE（Trace Buffer Extension）：每个 CPU 自带一个追踪缓冲区，直接写到该 CPU 的内存，不需要外部 funnel/ETR。这是 ARMv8.4 引入的简化路径，驱动仍复用 CoreSight 的 sink 抽象与 perf AUX 机制。

## 配置与调试

| 配置项 | 本树状态 | 影响 |
| :-- | :-- | :-- |
| `CONFIG_HAVE_HW_BREAKPOINT` | y | `hw_breakpoint.o` 编译，ptrace 硬件断点可用 |
| `CONFIG_PERF_EVENTS` | y | perf 断点事件、CoreSight PMU 基础 |
| `CONFIG_KGDB` | **未启用** | `kgdb.o` 不编译，无内核远程调试 |
| `CONFIG_KPROBES` | 未启用 | 无动态插桩（BRK hook 链仍用于 BUG/WARN） |
| `CONFIG_FTRACE` | 未启用 | 无函数追踪 |
| `CONFIG_CORESIGHT` | **未启用** | ETM/TMC/TRBE 驱动不编译 |
| `CONFIG_DEBUG_INFO` | y | 带 DWARF 调试信息，gdb 可解析符号 |
| `CONFIG_ARM64_PTR_AUTH_KERNEL` | y | 内核返回地址带 PAC，unwind 需 strip |

调试切入点：

- **启动日志**：`found %d breakpoint and %d watchpoint registers.` 确认硬件槽位数。
- **`/sys/kernel/debug/debug_enabled`**：写 0 关闭自托管调试，让 JTAG 接管。
- **启动参数 `nodebugmon`**：同上，早期生效。
- **gdb**：`hbreak`/`watch` 走 `NT_ARM_HW_BREAK`/`NT_ARM_HW_WATCH`；`stepi` 走 `PTRACE_SINGLESTEP`。
- **perf**：`perf record -e mem:<addr>:x` 使用硬件断点 PMU；`perf record -e cs_etm//` 使用 CoreSight（需启用配置）。
- **BRK 立即数**：oops 里的 `brk #0x...` 可区分 BUG（`BUG_BRK_IMM`）、kgdb 动态断点、kprobe 等。
