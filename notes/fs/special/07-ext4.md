# ext4 — 第四代扩展文件系统

## 1. 概述与实现机制

ext4 是 Linux 最广泛使用的磁盘文件系统，是 ext3 的演进版本。支持最大 1EB 文件系统和 16TB 单个文件。

### 核心特性

- **日志 (Journal)**：元数据事务日志，保证崩溃一致性
- **延迟分配 (Delayed Allocation)**：写回时批量分配磁盘块
- **多块分配 (Multiblock Allocator)**：一次分配多个连续块
- **extents**：用 extent 树替代传统 block map 管理大文件
- **Htree 目录索引**：通过 B-Tree 变体加速大目录查找
- **在线调整大小**：文件系统挂载时扩展
- **纳秒时间戳**：支持纳秒级时间精度
- **预分配**：fallocate() 系统调用支持

### 磁盘布局

```
磁盘块组 (Block Group) 0:
┌──────────┬──────────────┬──────────────┬──────────────┬──────────────────┐
│ Super    │ Group        │ Data Block   │ Inode        │ Data Blocks      │
│ Block    │ Descriptors  │ Bitmap       │ Table        │ (文件内容)       │
├──────────┼──────────────┼──────────────┼──────────────┼──────────────────┤
│ 块组 0   │ 块组 0       │ 块组 0       │ 块组 0       │ 块组 0           │
└──────────┴──────────────┴──────────────┴──────────────┴──────────────────┘

磁盘块组 (Block Group) N:
┌──────────┬──────────────┬──────────────┬──────────────┬──────────────────┐
│ Super    │ Group        │ Data Block   │ Inode        │ Data Blocks      │
│ Block    │ Descriptors  │ Bitmap       │ Table        │ (文件内容)       │
│ (备份)   │ (备份)       │              │              │                  │
└──────────┴──────────────┴──────────────┴──────────────┴──────────────────┘

每个块组包含:
  - super_block: 文件系统全局信息 (备份)
  - group descriptors: 块组描述符表
  - data block bitmap: 数据块使用位图 (1 bit/块)
  - inode bitmap: inode 使用位图
  - inode table: inode 数组 (每个 inode 256 字节)
  - data blocks: 实际文件数据
```

### 2.3 ext4_extent — extent 结构

```c
// 文件: fs/ext4/ext4.h
struct ext4_extent {
    __le32  ee_block;    // 逻辑块号 (起始)
    __le16  ee_len;      // extent 长度 (块数)
    __le16  ee_start_hi; // 物理块号高 16 位
    __le32  ee_start_lo; // 物理块号低 32 位
};
// 单个 extent 可映射最多 32768 个连续块 (128MB, 4K 块大小)

struct ext4_extent_header {
    __le16  eh_magic;    // 魔数 (0xF30A)
    __le16  eh_entries;  // 有效条目数
    __le16  eh_max;      // 最大条目数
    __le16  eh_depth;    // 树深度 (0=叶子)
    __le32  eh_generation; // 生成号
};

struct ext4_extent_idx {
    __le32  ei_block;    // 索引覆盖的逻辑块号
    __le32  ei_leaf_lo;  // 叶子节点块号低 32 位
    __le16  ei_leaf_hi;  // 叶子节点块号高 16 位
    __u16   ei_unused;
};
```

### 2.4 ext4_group_desc — 块组描述符

```c
// 文件: fs/ext4/ext4.h
struct ext4_group_desc {
    __le32  bg_block_bitmap_lo;     // 块位图低 32 位
    __le32  bg_inode_bitmap_lo;     // inode 位图低 32 位
    __le32  bg_inode_table_lo;      // inode 表起始块低 32 位
    __le16  bg_free_blocks_count_lo; // 空闲块数低 16 位
    __le16  bg_free_inodes_count_lo; // 空闲 inode 数低 16 位
    __le16  bg_used_dirs_count_lo;   // 目录数低 16 位
    __le16  bg_flags;               // 标志位
    __le32  bg_exclude_bitmap_lo;   // 排除位图低 32 位
    __le16  bg_block_bitmap_csum_lo; // 块位图校验和
    __le16  bg_inode_bitmap_csum_lo; // inode 位图校验和
    __le16  bg_itable_unused_lo;     // 未使用 inode 表项数
    __le16  bg_checksum;            // 校验和 (低 16 位)
    __le32  bg_block_bitmap_hi;     // 块位图高 32 位
    __le32  bg_inode_bitmap_hi;     // inode 位图高 32 位
    __le32  bg_inode_table_hi;      // inode 表高 32 位
    __le16  bg_free_blocks_count_hi; // 空闲块数高 16 位
    __le16  bg_free_inodes_count_hi; // 空闲 inode 数高 16 位
    __le16  bg_used_dirs_count_hi;   // 目录数高 16 位
    __le16  bg_itable_unused_hi;     // 未使用 inode 表项数
    __le32  bg_exclude_bitmap_hi;   // 排除位图高 32 位
    __le16  bg_block_bitmap_csum_hi; // 块位图校验和高 16 位
    __le16  bg_inode_bitmap_csum_hi; // inode 位图校验和高 16 位
    __le32  bg_reserved;            // 保留
};
```

## 3. API 与使用方法

### 3.1 用户空间使用

```bash
# 创建 ext4 文件系统
mkfs.ext4 /dev/sdb1

# 挂载
mount -t ext4 /dev/sdb1 /mnt/data

# 挂载选项示例
mount -t ext4 -o noatime,data=ordered,delalloc /dev/sdb1 /mnt/data

# 调整文件系统
resize2fs /dev/sdb1            # 在线调整大小
tune2fs -l /dev/sdb1           # 查看文件系统信息
e2fsck /dev/sdb1               # 检查文件系统
```

### 3.2 内核内部 API

```c
#include <linux/ext4_fs.h>

// 核心操作
struct inode *ext4_iget(struct super_block *sb, unsigned long ino);
int ext4_write_inode(struct inode *inode, struct writeback_control *wbc);
void ext4_evict_inode(struct inode *inode);

// 块分配
int ext4_map_blocks(handle_t *handle, struct inode *inode,
                    struct ext4_map_blocks *map, int flags);
int ext4_alloc_blocks(handle_t *handle, struct inode *inode, ...);

// 目录操作
struct inode *ext4_lookup(struct inode *dir, struct dentry *dentry, unsigned int flags);
int ext4_mkdir(struct mnt_idmap *idmap, struct inode *dir,
               struct dentry *dentry, umode_t mode);
int ext4_rmdir(struct inode *dir, struct dentry *dentry);

// 文件操作
ssize_t ext4_file_write_iter(struct kiocb *iocb, struct iov_iter *from);
ssize_t ext4_file_read_iter(struct kiocb *iocb, struct iov_iter *to);
int ext4_file_open(struct inode *inode, struct file *file);

// 日志操作
handle_t *ext4_journal_start_sb(struct super_block *sb, int nblocks);
int ext4_journal_stop(handle_t *handle);
```

## 4. 函数调用栈

### 4.1 挂载流程

```
mount -t ext4 /dev/sdb1 /mnt
  ↓ sys_mount() → do_new_mount() → vfs_get_tree()
    → ext4_init_fs_context(fc)                    // 初始化文件系统上下文
    → ext4_fill_super(sb, fc, false)              // 填充超级块 (核心)
      → ext4_read_super_block(sb, ...)            // 读取磁盘超级块
        → sb_bread(sb, block)                     // 从块设备读取
        → ext4_validate_super_block(sb)           // 校验超级块
      → ext4_es_init() / ext4_init_journal()      // 初始化 extent 状态/日志
      → ext4_setup_super(sb, es, ...)             // 设置超级块
      → ext4_load_journal(sb, es, ...)            // 加载日志
        → ext4_get_journal_inode(sb, ...)         // 获取日志 inode
        → jbd2_journal_load(sb->s_journal)        // 加载并恢复日志
      → ext4_group_desc_init(...)                 // 初始化块组描述符
      → ext4_set_root_inode(sb, ...)              // 设置根 inode
      → sb->s_root = d_make_root(root_inode)      // 创建根 dentry
```

### 4.2 文件写入流程

```
write(fd, buf, len)
  ↓ vfs_write() → ext4_file_write_iter()
    → ext4_buffered_write_iter(iocb, from)         // 缓冲写入
      → generic_perform_write(iocb, from)           // 通用写入循环
        → ext4_write_begin(file, pos, len, &folio, &fsdata)
          → ext4_da_write_begin()                   // 延迟分配模式
            → ext4_da_reserve_space(inode, ...)     // 预留空间 (不分配物理块)
            → block_write_begin()                   // 准备页缓存
              → grab_cache_page_write_begin()       // 获取/创建页缓存页
              → __block_write_begin()               // 准备 buffer_head
        → copy_page_from_iter_atomic()              // 拷贝用户数据到页缓存
        → ext4_write_end(file, pos, len, copied, folio, fsdata)
          → ext4_da_write_end()                     // 延迟分配结束
            → ext4_da_reserve_metadata()            // 预留元数据空间
            → mark_buffer_dirty()                   // 标记脏页
    ↓
    // 回写时 (pdflush/kswapd 或 sync):
    → ext4_writepages(mapping, wbc)                 // 批量写回
      → ext4_da_writepages_trans_blocks()           // 计算事务块数
      → ext4_journal_start()                        // 开始日志事务
      → ext4_da_write_pages()                       // 写入脏页
        → mpage_map_and_submit_extent()             // 映射并提交 extent
          → ext4_map_blocks(handle, inode, map, flags)  // 分配物理块
            → ext4_mb_new_blocks(handle, &ar, ...)  // 多块分配器
              → ext4_mb_normalize_request()         // 标准化请求
              → ext4_mb_regular_allocator()         // 常规分配
              → ext4_mb_find_by_goal()              // 按目标查找
              → ext4_mb_simple_scan()               // 简单扫描
          → mpage_submit_extent(handle, mpd, io)    // 提交 I/O
            → bio_alloc() / submit_bio()            // 创建并提交 bio
```

### 4.3 目录查找 (Htree)

```
ext4_lookup(inode, dentry, flags)
  → ext4_find_entry(inode, &dentry->d_name, &de, &res_dir)
    → ext4_dx_find_entry(inode, fname, ...)         // Htree 索引查找
      → ext4_dx_read_root(inode, ...)               // 读取根节点
      → ext4_dx_find_root(inode, ...)               // 查找根节点中的条目
        → dx_probe(fname, inode, &frame, ...)       // 探测 Htree 路径
          → ext4_bread(..., dx_root->block, ...)    // 读取索引块
          → dx_get_hash() 比较哈希值
          → dx_get_block() 获取下一级块号
        → while (frame->bh)                         // 逐层遍历
          → ext4_bread(..., block, ...)             // 读取索引块
          → dx_move_to_next()                       // 移动到下一级
      → ext4_search_dir(bh, fname)                  // 在叶子块中搜索
        → ext4_match(inode, fname, de)              // 匹配目录项
          → ext4_match_dentry()                     // dentry 匹配
    → ext4_get_inode(sb, ino)                       // 获取 inode
```

## 5. 流程图

### 5.1 ext4 写操作路径

```
用户 write(fd, buf, len)
    │
    ▼
ext4_file_write_iter()
    │
    ├── O_DIRECT? → ext4_dio_write_iter()
    │                  → ext4_dio_write_checks()
    │                  → iomap_dio_iomap_begin()
    │                  → ext4_iomap_begin() → ext4_map_blocks()
    │                  → __iomap_dio_rw()
    │
    └── 缓冲写入 → ext4_buffered_write_iter()
           │
           ▼
    generic_perform_write()
         │
         ├── ext4_write_begin()
         │     → ext4_da_write_begin()
         │       → ext4_da_reserve_space()  # 预留空间
         │       → grab_cache_page_write_begin()  # 获取页缓存
         │
         ├── copy_page_from_iter_atomic()  # 拷贝数据
         │
         └── ext4_write_end()
               → ext4_da_write_end()
                 → mark_buffer_dirty()  # 标记脏页
    │
    ▼
    回写触发 (sync/kswapd/内存压力)
         │
         ▼
    ext4_writepages()
         │
         ├── ext4_journal_start()  # 开始日志事务
         │
         ├── mpage_map_and_submit_extent()
         │     │
         │     ├── ext4_map_blocks()  # 分配物理块
         │     │     → ext4_mb_new_blocks()  # 多块分配器
         │     │
         │     └── mpage_submit_extent()
         │           → submit_bio()  # 提交 I/O 请求
         │
         ├── ext4_journal_stop()  # 结束日志事务
         │
         └── 日志提交 (jbd2)
               → jbd2_journal_commit_transaction()
```

## 6. 使用场景

| 场景 | 描述 | 示例 |
|------|------|------|
| **根文件系统** | 大多数 Linux 发行版的默认根文件系统 | `/` 挂载为 ext4 |
| **数据存储** | 数据库、文件服务器数据盘 | `/data` 挂载 ext4 |
| **大文件存储** | 视频监控、日志存储 | 单个文件可达 16TB |
| **嵌入式系统** | 需要日志保护的嵌入式设备 | 确保断电一致性 |
| **虚拟化存储** | KVM/QEMU 虚拟机磁盘镜像 | `qcow2` 格式底层使用 |
